config system global
    set hostname "FGT-01"
end

config system interface
    edit "port1"
        set ip 192.168.1.1 255.255.255.0
        set allowaccess ping https ssh http
    next
end

config firewall policy
    edit 1
        set name "Allow-Web"
        set srcintf "port1"
        set dstintf "port2"
        set action accept
        set service "HTTP" "HTTPS"
    next
end

config system admin
    edit "admin"
        set password "example-password"
    next
end
